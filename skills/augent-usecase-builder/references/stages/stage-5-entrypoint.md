# Stage 5: Entry Point and API

Wires the use case into a runnable application with Orchestrator.

## 5.1 The main.py pattern

In `usecases/<name>/ai_agents/main.py`:

```python
from __future__ import annotations

import asyncio
import os
import sys
from datetime import datetime, timezone
from pathlib import Path

from dotenv import load_dotenv

# MUST be called before framework imports and before Orchestrator usage
from ai_agents_core.logging import setup_logging, get_logger

# Load .env
load_dotenv()

# Path setup — ensures ai_agents_core is importable
core_path = str(Path(__file__).parent.parent.parent.parent)
if core_path not in sys.path:
    sys.path.append(core_path)

# Initialize logging BEFORE any framework imports
setup_logging(usecase_name="<usecase_name>")

# Framework imports — only after setup_logging
from ai_agents_core.orchestration import Orchestrator, EventBus, EventType, WorkflowEvent
from ai_agents_core.observability.adapters import EventBusAdapter

# Use-case imports
from .graph import build_graph

logger = get_logger("<usecase_name>.main")


async def main():
    """Entry point for <UseCase> workflow."""
    workflow_id = None  # auto-generated by Orchestrator

    # 1. Compile the graph
    graph = build_graph()

    # 2. Set up event bus for observability
    bus = EventBus()
    # Optional: attach Langfuse adapter
    # adapter = EventBusAdapter(langfuse_instance)
    # await bus.subscribe(EventType.WORKFLOW_COMPLETED, adapter.handle_event)

    # 3. Create orchestrator
    orchestrator = Orchestrator(
        graph=graph,
        event_bus=bus,
        workflow_timeout_s=3600,  # 1 hour — adjust per use case
    )

    # 4. Define initial input
    initial_input = {
        "task": "Your task description here",
        # ... other initial state fields
    }

    # 5. Execute
    logger.info("Starting workflow")
    result = await orchestrator.run(initial_input=initial_input)

    # 6. Report
    logger.info(
        "Workflow complete — cycle_state=%s processed=%d discarded=%d errors=%d",
        result["cycle_state"],
        result.get("items_processed", 0),
        result.get("items_discarded", 0),
        len(result.get("errors", [])),
    )

    return result


if __name__ == "__main__":
    result = asyncio.run(main())
    print(f"Cycle state: {result['cycle_state']}")
    print(f"Items processed: {result.get('items_processed', 0)}")
```

## 5.2 Path setup explanation

The `sys.path.append()` ensures the use case can find `ai_agents_core` at the
project root. The path math is:
```
usecases/<name>/ai_agents/main.py
→ ../../ → usecases/<name>/
→ ../../../ → usecases/
→ ../../../../ → project root (where ai_agents_core/ lives)
```

## 5.3 Adding a FastAPI endpoint (optional)

If the use case needs a REST API:

```python
# usecases/<name>/ai_agents/api/main.py
from __future__ import annotations

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI(title="<UseCase> API", version="1.0.0")


class RunRequest(BaseModel):
    task: str


class RunResponse(BaseModel):
    workflow_id: str
    cycle_state: str
    items_processed: int


@app.post("/api/v1/run", response_model=RunResponse)
async def run_workflow(request: RunRequest):
    """Trigger a workflow run."""
    from usecases.<name>.ai_agents.graph import build_graph
    from ai_agents_core.orchestration import Orchestrator

    graph = build_graph()
    orchestrator = Orchestrator(graph=graph)

    result = await orchestrator.run(
        initial_input={"task": request.task}
    )

    return RunResponse(
        workflow_id=result["workflow_id"],
        cycle_state=result["cycle_state"].value,
        items_processed=result.get("items_processed", 0),
    )


@app.get("/api/v1/health")
async def health():
    return {"status": "healthy"}
```

**API directory structure:**
```
usecases/<name>/ai_agents/api/
├── __init__.py
├── main.py          # FastAPI app
└── models.py        # Request/response Pydantic models
```

## 5.5 Adding Langfuse observability

```python
from langfuse import Langfuse
from ai_agents_core.observability.adapters import EventBusAdapter

langfuse = Langfuse(
    public_key=os.getenv("LANGFUSE_PUBLIC_KEY"),
    secret_key=os.getenv("LANGFUSE_SECRET_KEY"),
    host=os.getenv("LANGFUSE_HOST"),
)

bus = EventBus()
adapter = EventBusAdapter(langfuse)
await bus.subscribe(EventType.WORKFLOW_STARTED, adapter.on_workflow_started)
await bus.subscribe(EventType.WORKFLOW_COMPLETED, adapter.on_workflow_completed)
await bus.subscribe(EventType.WORKFLOW_FAILED, adapter.on_workflow_failed)
await bus.subscribe(EventType.AGENT_STARTED, adapter.on_agent_started)
await bus.subscribe(EventType.AGENT_COMPLETED, adapter.on_agent_completed)
await bus.subscribe(EventType.LLM_STARTED, adapter.on_llm_started)
await bus.subscribe(EventType.LLM_COMPLETED, adapter.on_llm_completed)

orchestrator = Orchestrator(graph=graph, event_bus=bus)
```
